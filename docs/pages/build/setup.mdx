import React, { useEffect, useMemo, useState } from "react";

// Simple PWA-ready Daily Schedule App
// - Saves tasks to localStorage
// - Lets you add/remove tasks
// - Basic reminder (browser notification if enabled)
// - Installable on phone when deployed (see instructions below)

export default function DailyScheduleApp() {
  const [tasks, setTasks] = useState(() => {
    const saved = localStorage.getItem("tasks");
    return (
      JSON.parse(saved) || [
        { id: 1, time: "06:30", task: "Wake up" },
        { id: 2, time: "07:30", task: "Learning (finance/trading)" },
        { id: 3, time: "09:30", task: "Practice / Job search" },
        { id: 4, time: "12:30", task: "Lunch / Break" },
        { id: 5, time: "13:30", task: "Skill building" },
        { id: 6, time: "15:30", task: "Exercise / Walk" },
        { id: 7, time: "17:00", task: "Free time" },
        { id: 8, time: "18:30", task: "Review + learning" },
        { id: 9, time: "20:00", task: "Relax" },
        { id: 10, time: "22:30", task: "Sleep" }
      ]
    );
  });

  const [newTime, setNewTime] = useState("");
  const [newTask, setNewTask] = useState("");
  const [permission, setPermission] = useState("default");

  useEffect(() => {
    localStorage.setItem("tasks", JSON.stringify(tasks));
  }, [tasks]);

  useEffect(() => {
    if ("Notification" in window) {
      setPermission(Notification.permission);
    }
  }, []);

  const sortedTasks = useMemo(() => {
    return [...tasks].sort((a, b) => a.time.localeCompare(b.time));
  }, [tasks]);

  const addTask = () => {
    if (newTime && newTask) {
      setTasks([
        ...tasks,
        { id: Date.now(), time: newTime, task: newTask }
      ]);
      setNewTime("");
      setNewTask("");
    }
  };

  const deleteTask = (id) => {
    setTasks(tasks.filter((t) => t.id !== id));
  };

  const requestNotifications = async () => {
    if (!("Notification" in window)) return;
    const res = await Notification.requestPermission();
    setPermission(res);
  };

  // Very simple checker (runs every minute)
  useEffect(() => {
    if (!("Notification" in window) || permission !== "granted") return;

    const interval = setInterval(() => {
      const now = new Date();
      const hh = String(now.getHours()).padStart(2, "0");
      const mm = String(now.getMinutes()).padStart(2, "0");
      const current = `${hh}:${mm}`;

      tasks.forEach((t) => {
        if (t.time === current) {
          new Notification("Task Reminder", { body: t.task });
        }
      });
    }, 60000);

    return () => clearInterval(interval);
  }, [tasks, permission]);

  return (
    <div className="p-4 max-w-md mx-auto">
      <h1 className="text-xl font-bold mb-4">Daily Schedule</h1>

      <button
        onClick={requestNotifications}
        className="mb-3 bg-green-500 text-white p-2 rounded w-full"
      >
        Enable Reminders
      </button>

      <ul className="mb-4">
        {sortedTasks.map((item) => (
          <li
            key={item.id}
            className="flex justify-between items-center border-b py-2"
          >
            <span className="font-mono">{item.time}</span>
            <span className="flex-1 mx-2">{item.task}</span>
            <button
              onClick={() => deleteTask(item.id)}
              className="text-red-500"
            >
              ✕
            </button>
          </li>
        ))}
      </ul>

      <div className="flex flex-col gap-2">
        <input
          type="time"
          value={newTime}
          onChange={(e) => setNewTime(e.target.value)}
          className="border p-2 rounded"
        />

        <input
          type="text"
          placeholder="Task"
          value={newTask}
          onChange={(e) => setNewTask(e.target.value)}
          className="border p-2 rounded"
        />

        <button
          onClick={addTask}
          className="bg-blue-500 text-white p-2 rounded"
        >
          Add Task
        </button>
      </div>

      <p className="text-xs text-gray-500 mt-4">
        Tip: After deploying, open in Chrome → "Add to Home Screen" to install.
      </p>
    </div>
  );
}
